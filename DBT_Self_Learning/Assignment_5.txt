(declare, open, fetch, close)
==========================================================================================

delimiter //

create procedure cursordemo()
begin
declare v_empid int;
declare v_empname varchar(50);
declare done int default 0;

declare emp_cursor cursor for
select empid,empname from employee;

declare continue handler for not found
set done=1;

open emp_cursor;

read_loop: loop

fetch emp_cursor into v_empid,v_empname;

if done=1 then
leave read_loop;
end if;

select v_empid,v_empname;

end loop;

close emp_cursor;

end //

delimiter ;

delimiter //

create procedure salarycursor()
begin
declare v_name varchar(50);
declare v_salary int;
declare done int default 0;

declare cur cursor for
select empname,salary from employee;

declare continue handler for not found
set done=1;

open cur;

read_loop: loop

fetch cur into v_name,v_salary;

if done=1 then
leave read_loop;
end if;

select v_name,v_salary;

end loop;

close cur;

end //
