delimiter //

select 'high salary' as result;

delimiter //

create procedure departmentname(in did int)
begin
case did
when 1 then select 'it';
when 2 then select 'hr';
when 3 then select 'finance';
else select 'unknown';
end case;
end //

delimiter //

create procedure printnumbers()
begin
declare i int default 1;
loop_label: loop
select i;
set i=i+1;
if i>5 then
leave loop_label;
end if;
end loop;
end //
