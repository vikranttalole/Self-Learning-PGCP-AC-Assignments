Question: 7) Exception handling in Stored Procedure Exception handling with various handler actions
===========================================================================================================
1. continue handler
=========================================================================================

delimiter //

create procedure continuehandlerdemo()
begin
declare continue handler for sqlexception
select 'error occurred but execution continued';

insert into employee values (101,'rahul',40000);

select 'procedure completed';
end //

delimiter ;

mysql> call continuehandlerdemo();

2. exit handler
=========================================================================================

delimiter //

create procedure exithandlerdemo()
begin
declare exit handler for sqlexception
select 'error occurred and procedure terminated';

insert into employee values (101,'sneha',60000);

end //

delimiter ;

3. not found handler
=========================================================================================
delimiter //

create procedure notfounddemo()
begin
declare v_name varchar(50);

declare continue handler for not found
select 'record not found';

select empname into v_name from employee where empid=999;

select v_name;
end //

delimiter ;

mysql> call notfounddemo();
